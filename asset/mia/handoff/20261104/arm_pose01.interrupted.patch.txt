+def _from_to(a: np.ndarray, b: np.ndarray) -> np.ndarray:
+    """Proper rotation from a measured rest segment to an absolute target."""
+    a = np.asarray(a, float); a /= np.linalg.norm(a); b = np.asarray(b, float); b /= np.linalg.norm(b)
+    v = np.cross(a, b); c = float(a @ b); k = np.array(((0., -v[2], v[1]), (v[2], 0., -v[0]), (-v[1], v[0], 0.)))
+    if c < -.99999:  # deterministic perpendicular axis for the only singular case
+        axis = np.cross(a, (1., 0., 0.));
+        if np.linalg.norm(axis) < 1.e-6: axis = np.cross(a, (0., 1., 0.))
+        axis /= np.linalg.norm(axis); return 2*np.outer(axis, axis)-np.eye(3)
+    return np.eye(3) + k + k @ k / (1.+c)
